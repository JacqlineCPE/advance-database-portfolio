Q2: Monthly revenue for the last 12 months
------------------------------------------
EXPLAIN ANALYZE

       COUNT(*) AS orders,

Q3: Top 10 products by units sold in a date range
-------------------------------------------------

       SUM(i.quantity) AS units_sold,
       SUM(i.quantity * i.unit_price) AS revenue

JOIN customer_order o ON o.order_id = i.order_id
JOIN product p ON p.product_id = i.product_id
JOIN category c ON c.category_id = p.category_id
WHERE o.status = 'delivered'
  AND o.order_date BETWEEN '2026-01-01' AND '2026-06-30 23:59:59'

Q4: Products at or below reorder level
--------------------------------------
EXPLAIN ANALYZE
SELECT p.sku, p.name, inv.quantity_on_hand, inv.reorder_level,
       inv.reorder_level - inv.quantity_on_hand AS shortfall

Q5: Latest status of every in-transit delivery
----------------------------------------------

Q7: Customers ranked by spend within their state
------------------------------------------------
EXPLAIN ANALYZE
SELECT state, full_name, total_spent, state_rank
FROM (
    SELECT a.state, c.full_name,
           SUM(o.total_amount) AS total_spent,
           RANK() OVER (PARTITION BY a.state ORDER BY SUM(o.total_amount) DESC) AS state_rank
    FROM customer_order o
    JOIN customer c ON c.customer_id = o.customer_id
    JOIN address a ON a.address_id = o.address_id
    WHERE o.status = 'delivered'
    GROUP BY a.state, c.customer_id, c.full_name
) ranked
WHERE state_rank <= 3
ORDER BY state, state_rank;

-> Sort: ranked.state, ranked.state_rank  (cost=2.6..2.6 rows=0) (actual time=1111..1111 rows=24 loops=1)

Q5b: Q5 rewritten with a window function instead of a correlated subquery
-------------------------------------------------------------------------
EXPLAIN ANALYZE
SELECT delivery_id, tracking_number, courier_name, latest_status, location, recorded_at
FROM (
    SELECT d.delivery_id, d.tracking_number, d.courier_name,
           h.status AS latest_status, h.location, h.recorded_at,
           ROW_NUMBER() OVER (PARTITION BY h.delivery_id ORDER BY h.recorded_at DESC) AS rn
    FROM delivery d
    JOIN delivery_status_history h ON h.delivery_id = d.delivery_id
    WHERE d.status = 'in_transit'
) latest
WHERE rn = 1
ORDER BY recorded_at DESC
LIMIT 50;

-> Limit: 50 row(s)  (cost=3.6..3.6 rows=0) (actual time=1009..1009 rows=50 loops=1)
    -> Sort: latest.recorded_at DESC, limit input to 50 row(s) per chunk  (cost=3.6..3.6 rows=0) (actual time=1009..1009 rows=50 loops=1)
        -> Index lookup on latest using <auto_key0> (rn=1)  (cost=0.35..3.5 rows=10) (actual time=987..998 rows=15001 loops=1)
            -> Materialize  (cost=0..0 rows=0) (actual time=987..987 rows=45003 loops=1)
                -> Window aggregate: row_number() OVER (PARTITION BY h.delivery_id ORDER BY h.recorded_at desc )   (actual time=716..824 rows=45003 loops=1)
                    -> Sort: h.delivery_id, h.recorded_at DESC  (actual time=716..731 rows=45003 loops=1)
